CREATE DATABASE cesc;
USE cesc;

-- user
CREATE TABLE users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    user_type ENUM('USER', 'SPONSOR', 'MAINTAINER') NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- charging station
CREATE TABLE charging_stations (
    station_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    location VARCHAR(255) NOT NULL,
    capacity INT NOT NULL,
    available_slots INT NOT NULL
);

-- scooters
CREATE TABLE scooters (
    scooter_id INT AUTO_INCREMENT PRIMARY KEY,
    sponsor_user_id INT,
    vehicle_number VARCHAR(50) UNIQUE NOT NULL,
    make VARCHAR(50),
    model VARCHAR(50),
    color VARCHAR(30),
    battery_capacity INT,
    current_charge_level INT,
    status ENUM('AVAILABLE','IN_USE','CHARGING','MAINTENANCE') DEFAULT 'AVAILABLE',
    current_station_id INT,
    FOREIGN KEY (sponsor_user_id) REFERENCES users(user_id),
    FOREIGN KEY (current_station_id) REFERENCES charging_stations(station_id)
);

CREATE TABLE maintenance_alerts (
    alert_id INT AUTO_INCREMENT PRIMARY KEY,
    scooter_id INT,
    alert_type VARCHAR(50),
    notes VARCHAR(255),
    status ENUM('OPEN','RESOLVED') DEFAULT 'OPEN',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (scooter_id) REFERENCES scooters(scooter_id)
);

CREATE TABLE maintenance_tasks (
    task_id INT AUTO_INCREMENT PRIMARY KEY,
    scooter_id INT,
    maintainer_user_id INT,
    description VARCHAR(255),
    status ENUM('PENDING','COMPLETED') DEFAULT 'PENDING',
    due_date DATE,
    FOREIGN KEY (scooter_id) REFERENCES scooters(scooter_id),
    FOREIGN KEY (maintainer_user_id) REFERENCES users(user_id)
);

CREATE TABLE account_transactions (
    transaction_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT,
    activity_name VARCHAR(50),
    amount DOUBLE,
    transaction_type ENUM('CREDIT','DEBIT') NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(user_id)
);
